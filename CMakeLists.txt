cmake_minimum_required(VERSION 3.10)
project(CleanCitySimulator CXX)
set(CMAKE_CXX_STANDARD 11)
find_package(OpenGL REQUIRED)
find_package(GLUT REQUIRED)
file(GLOB SRC src/*.cpp)
add_executable(CleanCitySimulator ${SRC})
target_include_directories(CleanCitySimulator PRIVATE include ${GLUT_INCLUDE_DIR})
target_link_libraries(CleanCitySimulator ${OPENGL_LIBRARIES} ${GLUT_LIBRARIES})
